void checkForUpdates() {
  Serial.println("Checking for firmware updates...");

  WiFiClientSecure client;
  client.setInsecure();

  HTTPClient http;
  http.begin(client, VERSION_URL);

  int httpCode = http.GET();

  if (httpCode == HTTP_CODE_OK) {
    String latestVersion = http.getString();
    latestVersion.trim();

    if(latestVersion != FIRMWARE_VERSION) {
      Serial.println("New version available - starting update!");
      http.end();
      performOTAUpdate();
    } else {
      Serial.println("No new version available");
    }
  } else {
    Serial.println("Failed to check version");
  }
  http.end();
}

void performOTAUpdate() {
  WiFiClientSecure client;
  client.setInsecure();

  Serial.println("Starting OTA update");
  setAllLEDs(0, 100, 0);


  const esp_partition_t* update_partition = esp_ota_get_next_update_partition(NULL);

  httpUpdate.onStart([]() {
    Serial.println("OTA Update Started");
  });

  httpUpdate.onEnd([]() {
    Serial.println("OTA Update Finished");
  });

  httpUpdate.onError([](int error) {
    Serial.printf("Oops an error occured while performing the OTA update: %d\n", error);
    digitalWrite(48, HIGH);
  });

  t_httpUpdate_return ret = httpUpdate.update(client, FIRMWARE_URL);

  switch (ret) {
    case HTTP_UPDATE_FAILED:
      Serial.printf("Update failed with error %d: %s\n", httpUpdate.getLastError(), httpUpdate.getLastErrorString().c_str());
      digitalWrite(48, HIGH);
      break;
    
    case HTTP_UPDATE_NO_UPDATES:
      Serial.println("No new updates");
      break;

    case HTTP_UPDATE_OK:
      Serial.println("Update success! Rebooting...");
      delay(1000);
      ESP.restart();
      break;
  }
}

  */

  static unsigned long lastDisconnectLog = 0;
  if (!ap_mode && WiFi.status() != WL_CONNECTED) {
    if (millis() - lastDisconnectLog > 5000) {
      Serial.println("WiFi disconnected, auto-reconnecting...");
      digitalWrite(21, LOW);
      lastDisconnectLog = millis();
    }
  } else if (WiFi.status() == WL_CONNECTED) {
    digitalWrite(21, HIGH);  // LED on when connected
  }
  
  // Update LED colors if connected
  if (WiFi.status() == WL_CONNECTED && (millis() - lastColorUpdate >= colorUpdateInterval) && !safemode) {
    updateLEDColors();
    lastColorUpdate = millis();
  }

  delay(10);
  yield();
}

void checkBootButton() {
  static unsigned long bootButtonPressTime = 0;
  static bool bootButtonPressed = false;

  if (digitalRead(0) == LOW) {
    if (!bootButtonPressed) {
      bootButtonPressed = true;
      bootButtonPressTime = millis();
    } else if (millis() - bootButtonPressTime > 3000) {
      // Held for 3 seconds - wipe credentials and restart
      Serial.println("Reset button held - clearing WiFi credentials and restarting...");
      
      // Flash LEDs red to confirm reset
      setAllLEDs(255, 0, 0);
      delay(500);
      setAllLEDs(0, 0, 0);
      delay(500);
      setAllLEDs(255, 0, 0);
      delay(500);
      
      preferences.clear();
      delay(500);
      ESP.restart();
    }
  } else {
    bootButtonPressed = false;
  }
}

void increaseBrightness() {

      if (brightness + 2 <= 16) {
        brightness = brightness + 2;
        ws2812b1.setBrightness(brightness);
        ws2812b2.setBrightness(brightness);
        ws2812b3.setBrightness(brightness);

}

void switchMode() {

  static bool shortPressHandled = false;

  if (digitalRead(42) == LOW) {
    if(!modeButtonPressed) {
      modeButtonPressed = true;
      modeButtonPressTime = millis();
      modeToggled = false;
      shortPressHandled = false;
    } else if (millis() - modeButtonPressTime > 3000 && !modeToggled) {
      if (live) {
        digitalWrite(47, LOW);
        serverURL = serverURLbackup;
        live = false;
        modeToggled = true;
      } else {
        digitalWrite(47, HIGH);
        serverURL = serverURLlive;
        live = true;
        modeToggled = true;
      }
    }
  } else {
    if (modeButtonPressed && !modeToggled && !shortPressHandled) {
      unsigned long pressDuration = millis() - modeButtonPressTime;
      if (pressDuration > 50 && pressDuration < 3000) {
        mode = !mode;
        shortPressHandled = true;
        updateLEDColors();
      }
    }
    modeButtonPressed = false;
    shortPressHandled = false;
  }
}

void startAccessPoint() {
  Serial.println("Starting Access Point Mode...");
  ap_mode = true;
  
  WiFi.mode(WIFI_AP);
  WiFi.softAP(ap_ssid, ap_password);
  
  IPAddress IP = WiFi.softAPIP();
  Serial.print("AP IP address: ");
  Serial.println(IP);
  
  dnsServer.start(53, "*", IP);
  setupWebServer();

  digitalWrite(21, LOW);
  digitalWrite(47, LOW);
  
  Serial.println("Access Point started - Connect to 'ESP32-LED-Setup' to configure WiFi");
}

void setupWebServer() {
  server.on("/", HTTP_GET, handleRoot);
  server.on("/status", HTTP_GET, handleStatus);
  server.on("/connect", HTTP_POST, handleConnect);
  server.on("/clear", HTTP_POST, handleClear);
  server.onNotFound(handleRoot);
  
  server.begin();
  Serial.println("Web server started");
}

void handleRoot() {
  server.send(200, "text/html", html_page);
}

void handleStatus() {
  bool wifi_connected = (WiFi.status() == WL_CONNECTED);
  String json = "{\"ssid\":\"" + stored_ssid + "\",\"connected\":" + 
                (wifi_connected ? "true" : "false") + ",\"mac\":\"" + mac + "\"}";
  server.send(200, "application/json", json);
}

void handleConnect() {
  String ssid = server.arg("ssid");
  String password = server.arg("password");
  
  Serial.println("Received WiFi credentials:");
  Serial.println("SSID: " + ssid);
  
  preferences.putString("ssid", ssid);
  preferences.putString("password", password);
  stored_ssid = ssid;
  stored_password = password;
  
  if (connectToWiFi(ssid, password)) {
    server.send(200, "text/plain", "Success");
  } else {
    server.send(500, "text/plain", "Failed");
  }
}

void handleClear() {
  Serial.println("Clearing stored WiFi credentials...");
  preferences.clear();
  stored_ssid = "";
  stored_password = "";
  server.send(200, "text/plain", "Cleared");
  delay(1000);
  ESP.restart();
}

bool connectToWiFi(String ssid, String password) {
  Serial.println("Connecting to WiFi: " + ssid);

    if (ap_mode) {
    WiFi.mode(WIFI_AP_STA);
  } else {
    WiFi.mode(WIFI_STA);
  }
  
  WiFi.begin(ssid.c_str(), password.c_str());
  
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 20) {
    delay(500);
    Serial.print(".");
    attempts++;
    yield();
  }
  
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nWiFi Connected!");
    Serial.print("IP Address: ");
    Serial.println(WiFi.localIP());
    digitalWrite(21, HIGH);

    if (ap_mode) {
      Serial.println("Shutting down Access Point...");
      dnsServer.stop();
      WiFi.softAPdisconnect(true);  // Disconnect AP and disable
      delay(100);
      WiFi.mode(WIFI_STA);  // Switch to Station-only mode
      ap_mode = false;
      Serial.println("Access Point shut down - now in Station mode only");
    }

    return true;
  } else {
    Serial.println("\nWiFi Connection Failed");
    digitalWrite(21, LOW);
    digitalWrite(47, LOW);
    return false;
  }
}
